void solve(){
    int n, ans = 1; cin >> n;
    multiset<int>tow;
    for(int i = 0; i < n; i++){
        int x; cin >> x;
        auto it = tow.upper_bound(x);
        if(it!=tow.end()) tow.erase(it);
        tow.insert(x); 
    }
    cout << tow.size() << endl;
}



signed main(){

    cin.tie(nullptr);
    solve();
    return 0;
}
