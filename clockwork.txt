#include <bits/stdc++.h>
using namespace std;
#define MAXN 500005
#define int long long
#define ld long double
#define endl "\n"

bool solve(){
    int n, a[MAXN];
    cin >> n;
    for(int i = 1; i<=n; i++) cin >> a[i];
    for(int i = 1; i<=n; i++){
        if(a[i]<=max(i-1, n-i)*2) return false;
    }
    return true;
    //cout<<endl;
}

signed main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t; cin >> t;
    while(t--){
        if(solve()) cout << "YES" << endl;
        else cout << "NO" << endl;
    }
    return 0;
}
