#include <bits/stdc++.h>
 
using namespace std;
#define ll long long
#define ld long double
int main() {
    ios_base::sync_with_stdio(0);
    cin.tie(0); cout.tie(0);
    ll n, t;
    cin >> n >> t;
    vector<ll>a(n-1);
    for(int i = 0; i < n-1; i++){
        cin >> a[i];
    }
    ll pos = 1;
    ll cnt = 0;
    string ans = "NO";
    while(pos < t){
        pos+=a[cnt];
        cnt+=a[cnt];
        if(pos == t){
            ans = "YES";
            break;
        }
    }
    cout << ans;
}
