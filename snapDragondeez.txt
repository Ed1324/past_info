int main() {
    ios_base::sync_with_stdio(0);
    cin.tie(0); cout.tie(0);
    int n, b;
    cin >> n >> b;
    vector<int>R(n);
    vector<int>S(b);
    int ans {0};
    for(int i {0}; i < n; i++){
        cin >> R[i];
    }
    for(int i {0}; i < b; i++){
        cin >> S[i];
    }
    if(n > b){
        for(int i {0}; i < b; i++){
            for(int j {0}; j < n; j++){
                if(S[i] == R[j]){
                    ans++;
                }
            }
        }
    } else {
        for(int i {0}; i < n; i++){
            for(int j {0}; j < b; j++){
                if(R[i] == S[j]){
                    ans++;
                }
            }
        }
    }
    cout << ans << "\n";
}
