#include <bits/stdc++.h>
using namespace std;
#define MAXN 200005
#define int long long
#define ld long double
#define endl "\n"

void solve(){
    int n, x; cin >> n >> x;
    vector<pair<int, int>>a;
    for(int i = 1; i <= n; i++) {
        int num; cin >> num; a.push_back({num, i});
    }
    sort(a.begin(), a.end());
    for(int i = 0; i < n; i++){
        int l = i+1, r = n-1;
        while(l < r){
            int sum = a[i].first + a[l].first + a[r].first;
            if(x == sum) {
                cout << a[i].second << " " << a[l].second << " " << a[r].second;
                exit(0);
            }
            if(sum < x) l++;
            else r--;
        }
        
    }
    cout << "IMPOSSIBLE" << endl;
}



signed main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    solve();
    return 0;
}
