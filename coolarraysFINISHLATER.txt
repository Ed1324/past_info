const int MAX_N = 1e5 + 5;
const ll MOD = 1e9 + 7;
const ll INF = 1e9;
const ld EPS = 1e-9;


int main() {
    ios_base::sync_with_stdio(0);
    cin.tie(0); cout.tie(0);
    int t;
    cin >> t;
    while(t--){
        int n, m, k;
        cin >> n >> m >> k;
        vector<int>a(n);
        vector<int>b(m);
        for(int i {0}; i < n; i++){
            int num;
            cin >> num;
            if(k >= num){
                a.push_back(num);
            }
        }
        for(int i {0}; i < m; i++){
            int num;
            cin >> num;
            if(num <= k){
                b.push_back(num);
            }
        }
        
        int numsfroma = 0;
        int numsfromb = 0;
        
        bool found {false};
        bool inb {false};
        bool ina {false};
        for(int i {1}; i <= k; i++){
            found = false;
            for(int j {0}; j < b.size(); j++){
                if(b[j] == i){
                    inb = true;
                    break;
                }
            }
            for(int j {0}; j < a.size(); j++){
                if(a[j] == i){
                    ina = true;
                    break;
                }
            }
            
            if(numsfroma < k/2){
                for(int j {0}; j < a.size(); j++){
                    if(a[j] == i){
                        found = true;
                        numsfroma++;
                        break;
                    }
                }
            }
            if(found == false && numsfromb < k/2){
                for(int j {0}; j < b.size(); j++){
                    if(b[j] == i){
                        found = true;
                        numsfromb++;
                        break;
                    }
                }
            }
            if(numsfroma >= k/2 && inb == true && numsfroma > numsfromb){
                numsfroma--;
                numsfromb++;
                inb = false;
            } else if(numsfromb >= k/2 && ina == true && numsfroma < numsfromb){
                numsfroma++;
                numsfromb--;
                ina = false;
            }
            if(found == false){
                break;
            }
        }
        if(found){
            cout << "YES" << "\n";
        } else {
            cout << "NO" << "\n";
        }
    }
    
}
