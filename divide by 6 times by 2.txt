#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using ld = long double;

int main() {
    ios_base::sync_with_stdio(0);
    cin.tie(0); cout.tie(0);
    
    int tc = 1;
    cin >> tc;
    
    while(tc--){
        ll n;
        int counter2 {0}, counter3 {0};
        cin >> n;
        while(n % 2 == 0){
            n/=2;
            counter2++;
        }
        while(n % 3 == 0){
            n/=3;
            counter3++;
        }
        if(counter2 > counter3 || n != 1){
            cout << -1 << "\n";
        } else{
            cout << 2*counter3 - counter2 << "\n";
        }
    }
    return 0;
}
