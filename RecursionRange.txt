#include <bits/stdc++.h>
#define ll long long
#define ld long double
using namespace std;
#define MAXN 200005
#define endl "\n"

int recursion(int lb, int ub){
    if(lb > ub) return 0; // base case
    if(lb % 2 == 0) return lb + recursion(lb+1, ub);
    else return recursion(lb+1, ub);
}

int main()
{ 
   ios::sync_with_stdio(false);
   cin.tie(nullptr);
   cout << recursion(2, 6) << endl;
   return 0;
}
