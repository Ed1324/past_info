#include <bits/stdc++.h>
using namespace std;
#define MAXN 200005
#define int long long
#define ld long double
#define endl "\n"

void solve(){
    int n, ans = 0, p=0; cin >> n;
    map<int, int>a{{0,1}};
    
    for(int i = 0; i < n; i++){
        int num; cin >> num;
        p+=num;
        int rem = ((p%n) + n) % n;
        ans+=a[rem];
        a[rem]++;
    }
    cout << ans << endl;
}


signed main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    solve();
    return 0;
}
