#include <bits/stdc++.h>
#define ll long long
using namespace std;
int main()
{
    ll w, h, d;
    ll n;
    cin >> w  >> h >> d;
    cin >> n;
    
    ll wn = gcd(n, w);
    ll nw = n/wn;
    
    ll hn = gcd(nw, h);
    ll nh = nw/hn;
    
    ll dn = gcd(nh, d);
    ll nd = nh/dn;
    if(nd != 1){
        cout << -1;
        return 0;
    }
    cout << wn-1 << " " << hn-1 << " " << dn-1;
    return 0;
}
