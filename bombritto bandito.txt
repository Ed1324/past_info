#include <iostream>
#include <algorithm>
#include <vector> 
#include <string>
using namespace std;

int main() {
    int t, k;
    cin >> t;
    vector<int>nums;
    for(int i {0}; i < t; i++){
        for(int i {0}; i < 4; i++){
            cin >> k;
            nums.push_back(k);
        }
        if(abs(nums[2]) > nums[3]){
            while(nums[0] != nums[1]){
                if(nums[2] != 0){
                    nums[2]++;
                    nums[0]--;
                }
                if(nums[0] == nums[1]){
                    break;
                }
                if(nums[3] != 0){
                    nums[3]--;
                    nums[0]--;
                }
            }
        } else {
            while(nums[0] != nums[1]){
                if(nums[3] != 0){
                    nums[3]--;
                    nums[0]--;
                }
                if(nums[0] == nums[1]){
                    break;
                }
                if(nums[2] != 0){
                    nums[2]++;
                    nums[0]--;
                }
            }
        }

        
        cout << nums[2] << " " << nums[3] << "\n";
        nums.clear();
    }
    
    return 0;
}
