#include <bits/stdc++.h>

using namespace std;
#define ll long long
#define ld long double
int main() {
    ios_base::sync_with_stdio(0);
    cin.tie(0); cout.tie(0);
    string n;
    cin >> n;
    for(int i = 0; i < n.length(); i++){
        if(n[i] == '5'){
            n[i] = '4';
        }
        if(n[i] == '6'){
            n[i] = '3';
        }
        if(n[i] == '7'){
            n[i] = '2';
        }
        if(n[i] == '8'){
            n[i] = '1';
        }
        if(n[i] == '9' && i != 0){
            n[i] = '0';
        }
    }
    cout << n;
}
