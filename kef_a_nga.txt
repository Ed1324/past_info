#include <bits/stdc++.h>
using namespace std;
#define ar array
#define ll long long
#define ld long double

int main() {
    ios_base::sync_with_stdio(0);
    cin.tie(0); cout.tie(0);
    int tc;
    cin >> tc;
    int ans = 1;
    int highest {0};
    vector<int>nums(tc);
    
    for(int i {0}; i < tc; i++){
        cin >> nums[i];
    }
    
    for(int i {0}; i < tc; i++){
        if(highest < ans){
            highest = ans;
        }
        if(nums[i+1] >= nums[i]){
            ans++;
        } else {
            ans = 1;
        }
    }
    cout << highest << endl;
}
