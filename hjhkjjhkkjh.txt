void solve(){
    int n,ans=1;cin>>n;
    vector<int>mp(n+1);
    for(int i=1;i<=n;i++){
        int a;cin>>a;mp[a]=i;
    }
    for(int i=1;i<n;i++){
        if(mp[i]>mp[i+1]) ans++;
    }
    cout<<ans<<endl;
}

    cin.tie(nullptr);
    solve();
    return 0;
}
