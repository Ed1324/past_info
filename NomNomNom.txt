#include <bits/stdc++.h>
using namespace std;
#define MAXN 200005
#define int long long
#define endl "\n"

void solve(){
    int t, a[MAXN];
    cin >> t;
    for(int i =0;i < t;i++){
        cin >> a[i];
    }
    int prev = a[0], ans = 1;
    int total = 0;
    for(int i = 1; i < t; i++){
        total+=a[i];
        if(total >= prev) ans++, prev = total, total=0;
    }
    cout << ans;
}

signed main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    solve();
    return 0;
}
