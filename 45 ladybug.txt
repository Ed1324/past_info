#include <bits/stdc++.h>
using namespace std;
#define MAXN 200005
#define int long long
#define ld long double
#define endl "\n"

void solve(){
    int n, k, q;
    cin >> n >> k >> q;
    int a[MAXN], l, r;
    for(int i = 0; i < n; i++) cin >> a[i];
    sort(a, a+n);
    for(int i = 0; i < q; i++){
        int s; cin >> s;
        int ans = INT_MAX;
        for(int j = 0; j+k-1 <n; j++){
            l = a[j], r = a[j+k-1];
            int cost = (r - l) + min(abs(s - l), abs(s - r));
            ans = min(ans, cost);
        }
        cout <<ans<< " ";
    }
}



signed main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    cout << endl;
    solve();
    return 0;
}
