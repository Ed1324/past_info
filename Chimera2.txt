#include <bits/stdc++.h>
using namespace std;
#define MAXN 200005
#define int long long
#define endl "\n"

void solve(){
    int n, ans = 0, lane = 0, prev;
    string result = "";
    string s, t, targ;

    cin >> n >> s >> t >> targ;

    bool found = true;

    for(int i = 0; i < n; i++){
        if(s[i] != targ[i] && t[i] != targ[i]){
            found = false;
            break;
        }
    }
    for(int i = 0; i < n; i++){
        prev = lane;
        if(s[i] == targ[i] && t[i] == targ[i]) lane = prev;
        else if(s[i] == targ[i]) lane = 1;
        else if(t[i] == targ[i]) lane = 2;
        if(lane == prev) continue;
        else ans++;
    }

    if(found) cout << "SUCCESS" << endl << ans-1 << endl;
    else cout << "PLAN FOILED" << endl;
}

signed main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    cout << endl;
    solve();
    return 0;
}
