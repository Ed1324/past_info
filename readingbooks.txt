#include <bits/stdc++.h>
using namespace std;
#define MAXN 10000005
#define int long long
#define ld long double
#define endl "\n"
 

void solve(){
    int n;cin>>n;
    vector<int>v;
    int to=0;
    for(int i=0;i<n;i++){
        int a;cin>>a;
        v.push_back(a);
        to+=a;
    }
    sort(v.begin(),v.end());
    int mx=v[n-1];
    if(mx<to-mx){
        cout<<to<<endl;
    }
    else cout<<mx*2<<endl;
}   
 
 
 
signed main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    solve();
    return 0;
}
