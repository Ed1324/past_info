#include <bits/stdc++.h>
using namespace std;
#define MAXN 1005
#define int long long
#define ld long double
#define endl "\n"

void solve(){
    int n, ans = 0;
    cin >> n;
    int divisor = 1;
    int highest = 1;
    while(divisor < n/highest){
        if(n % divisor == 0) ans++, highest = divisor;
        divisor++;
    }
    int square = 0;
    if(pow(highest, 2) == n) square = 1;
    if(n == 1) cout << 1 << endl;
    else if(n == 2 || n == 3) cout << 2 << endl;
    else if(square == 0) cout << ans*2 << endl;
    else if(square == 1) cout << ans * 2 -1 << endl;

}



signed main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    cout << endl;
    int t;
    cin >> t;
    while(t--){
        solve();
    }
    return 0;
}
