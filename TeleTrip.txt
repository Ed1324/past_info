#include <bits/stdc++.h>

using namespace std;
#define ll long long
#define ld long double
int main() {
    ios_base::sync_with_stdio(0);
    cin.tie(0); cout.tie(0);
    ll n;
    cin >> n;
    string s;
    cin >> s;
    
    map<ll, ll>mp{{0, 0}};
    ll bal = 0;
    for(int i = 0; i < n; i++){
        if(s[i] == 'R'){
            bal++;
        }
        if(s[i] == 'L'){
            bal--;
        }
        if(s[i] == 'T'){
            bal = 0;
        }
        if(mp.count(bal) == 0){
            mp[bal] = 0;
        }
    }
    cout << mp.size();
}
