#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define ld long double
int main()
{
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int t;
    cin >> t;
    while(t--)
    {   
        int r, c;
        cin >> r >> c;
        string row;
        int counter {0};
        vector<int>nums;
        for(int i {0}; i < r; i++){
            cin >> row;
            for(auto a : row){
                if(a == '#'){
                    counter++;
                }
            }
            if(counter != 0)
                nums.push_back(counter);
            counter = 0;
        }
        for(auto a : nums){
            cout << a << " ";
        }
    }
}
