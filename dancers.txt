#include <bits/stdc++.h>
using namespace std;
#define MAXN 200005
#define int long long
#define endl "\n"

void solve(){
    int d, t, ans = 0;
    cin >> d >> t;
    int a[d+1] = {0};
    for(int i = 0; i < t; i++){
        int b, c;
        cin >> b >> c;
        if(a[b] == 0) ans++, a[c]++;
        else a[b]--, a[c]++;
    }
    cout << ans << endl;
}

signed main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    cout << endl;
    solve();
    return 0;
}
