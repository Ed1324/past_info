#include <bits/stdc++.h>
using namespace std;
#define MAXN 200005
#define int long long
#define ld long double
#define endl "\n"

void solve(){
    int n; cin >> n;
    int a[MAXN];
    int ans = LLONG_MIN;
    int p = 0, temp;
    for(int i = 0; i < n; i++) cin >> a[i];
    for(int i = 0; i < n; i++){
        p+=a[i];
        temp = p;
        if(p < 0) p = 0;
        ans = max(ans, temp);
    }
    cout << ans << endl;
}



signed main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    solve();
    return 0;
}
