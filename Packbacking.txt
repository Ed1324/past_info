#include <bits/stdc++.h>
using namespace std;
#define MAXN 200005
#define ll long long
ll n, d[MAXN];
int main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    cin >> n;
    vector<ll>arr(n, 0);
    ll tot = 0, maxn = 0;
    for(int i = 0; i < n; i++){
      ll a;
      cin >> a;
      arr[a]++;
      d[i] = a;
      tot++;
      maxn = max(maxn, d[i]);
    }
    ll curr = 0, pos = 0, res = 0;
    while(curr < maxn){
      curr++;
      if(curr >= d[pos]) tot--, pos++, arr[d[pos]]--, res++;
    }
    for(int i = pos; i < n; i++){
      if(curr > d[pos]) curr--;
      else curr++, res++;
    }
    cout << res;
    return 0;
}
