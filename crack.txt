#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define ld long double
int main()
{
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int t;
    cin >> t;
    while(t--)
    {
        ll n, k, x;
        
        cin >> n >> k >> x;
        ll max {0}, min {k*(k+1)/2};
        while(k--){
            if(max > x) break;
            max+=n;
            n--;
        }
        if(x >= min && x <= max){
            cout << "YES" << "\n";
        } else {
            cout << "NO" << "\n";
        }
    }
}
