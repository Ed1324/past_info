#include <iostream>
#include <algorithm>
#include <vector> 
#include <string>
using namespace std;

int main() {
    int t;
    cin >> t;
    for(int i {0}; i < t; i++){
        int counter {0};
        string num;
        cin >> num;
        int integer = num.length();
        bool positive {false};
        
        for (int i {integer-1}; i >= 0; i--){
            if(num[i] != '0'){
                positive = true;
            } else if(positive){
                counter++;
            }
        }
        cout <<  integer - (counter+1) << "\n";
    }
    return 0;
}
