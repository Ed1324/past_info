#include <bits/stdc++.h>
using namespace std;
#define MAXN 200005
#define int long long
#define ld long double
#define endl "\n"

void solve(){
    int n; cin >> n;
    int a[MAXN];
    for(int i = 0; i < n; i++) cin >> a[i];
    sort(a, a+n);
    int med = a[n/2];
    int ans = 0;
    for(int i = 0 ; i < n; i++){
        ans += abs(a[i]-med);
    }
    cout << ans << endl;
}



signed main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    solve();
    return 0;
}
