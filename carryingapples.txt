#include <bits/stdc++.h>
using namespace std;
#define MAXN 200005
#define int long long
#define ld long double
#define endl "\n"

void solve(){
    int n; cin >> n;
    int a[MAXN];
    for(int i = 0; i < n; i++) cin >> a[i];
    int ans = LLONG_MAX;
    int rem = 0, tot = 0;
    for(int i = 0; i < n; i++) tot+=a[i];
    int l = 0, r = 0;
    int prev = l;
    for(int i = 0; i < n; i++){
        l = 0;
        for(int j = i+1; j < n; j++){
            l = prev;
            l+=a[j], prev = l;
            r = tot-l;
            ans = min(ans, abs(r-l));
            for(int k = i+1; k < j; k++){
                l-=a[k];
                r = tot-l;
                ans = min(ans, abs(r-l));
            }
        }
    }
    cout << ans << endl;
}


signed main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    solve();
    return 0;
}
