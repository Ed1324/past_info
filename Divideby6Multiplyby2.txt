#include <bits/stdc++.h>
using namespace std;
#define MAXN 20005
#define int long long
#define endl "\n"

int solve(){
    int n, ans = 0; cin >> n;
    while(n >= 6){
        if(n%6==0)n/=6;
        else n*=2;
        ans++;
        if(n > INT_MAX) return -1;
    }
    if(n==2||n==4||n==5) return -1;
    else if(n == 3) return ans+2;
    return ans;
}

signed main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t; cin >> t;
    while(t--){
        cout << solve() << endl;
    }
    return 0;
}
